  <script>
    // Global variable to track current post ID
    let currentPostId = null;

    /**
     * Opens the comments overlay for a specific post
     * @param {number} postId - The ID of the post to show comments for
     */
    function openCommentsOverlay(postId) {
      currentPostId = postId;
      const overlay = document.getElementById('comments-overlay');
      
      // Show overlay with flex display for centering
      overlay.classList.remove('hidden');
      overlay.classList.add('flex');
      
      // Prevent body scrolling when overlay is open
      document.body.style.overflow = 'hidden';
      
      // Load comments for this post
      loadComments(postId);
    }

    /**
     * Closes the comments overlay
     */
    function closeCommentsOverlay() {
      const overlay = document.getElementById('comments-overlay');
      overlay.classList.add('hidden');
      overlay.classList.remove('flex');
      
      // Restore body scrolling
      document.body.style.overflow = 'auto';
      
      // Clear current post ID
      currentPostId = null;
      
      // Clear comment input
      document.getElementById('comment-input').value = '';
    }

    /**
     * Loads comments for a specific post via AJAX
     * @param {number} postId - The ID of the post to load comments for
     */
    async function loadComments(postId) {
      const container = document.getElementById('comments-container');
      
      try {
        const response = await fetch(`/comments/${postId}`);
        const data = await response.json();
        
        if (data.comments.length === 0) {
          container.innerHTML = `
            <div class="text-center text-gray-500 py-8">
              <svg class="w-12 h-12 mx-auto mb-2 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" 
                      d="M8 12h.01M12 12h.01M16 12h.01M21 12c0 4.418-4.03 8-9 8-1.575 0-3.05-.322-4.313-.894L3 20.25l1.318-3.947C3.488 15.042 3 13.574 3 12c0-4.418 4.03-8 9-8s9 3.582 9 8z">
                </path>
              </svg>
              <p>No comments yet. Be the first to comment!</p>
            </div>
          `;
        } else {
          container.innerHTML = data.comments.map(comment => `
            <div class="flex space-x-3 mb-4 p-3 bg-gray-50 rounded-lg">
              <img src="{{ url_for('static', filename='uploads/profile-pic.jpg') }}" 
                   alt="profile" class="w-8 h-8 rounded-full flex-shrink-0">
              <div class="flex-1">
                <p class="text-sm">
                  <span class="font-semibold text-gray-800">${comment.username}</span>
                  <span class="text-gray-600 ml-2">${comment.content}</span>
                </p>
                <p class="text-xs text-gray-500 mt-1">${comment.timestamp}</p>
              </div>
            </div>
          `).join('');
        }
      } catch (error) {
        console.error('Error loading comments:', error);
        container.innerHTML = `
          <div class="text-center text-red-500 py-8">
            <p>Error loading comments. Please try again.</p>
          </div>
        `;
      }
    }

    /**
     * Submits a new comment for the current post
     */
    async function submitComment() {
      const input = document.getElementById('comment-input');
      const content = input.value.trim();
      
      if (!content) {
        alert('Please enter a comment');
        return;
      }
      
      if (!currentPostId) {
        alert('No post selected');
        return;
      }
      
      try {
        const response = await fetch(`/comments/${currentPostId}`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({ content: content })
        });
        
        if (response.ok) {
          const data = await response.json();
          
          // Clear input
          input.value = '';
          
          // Reload comments to show the new one
          loadComments(currentPostId);
          
          // Update comment count in the post
          const countElement = document.getElementById(`comment-count-${currentPostId}`);
          if (countElement) {
            countElement.textContent = parseInt(countElement.textContent) + 1;
          }
          
        } else {
          const errorData = await response.json();
          alert(errorData.error || 'Error adding comment');
        }
      } catch (error) {
        console.error('Error submitting comment:', error);
        alert('Error submitting comment. Please try again.');
      }
    }

    /**
     * Handles Enter key press in comment input
     * @param {KeyboardEvent} event - The keyboard event
     */
    function handleCommentKeyPress(event) {
      if (event.key === 'Enter') {
        submitComment();
      }
    }

    // Close overlay when clicking outside the content area
    document.getElementById('comments-overlay').addEventListener('click', function(event) {
      if (event.target === this) {
        closeCommentsOverlay();
      }
    });
  </script>
