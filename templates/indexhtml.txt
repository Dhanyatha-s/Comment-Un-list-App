  <script>
    // Global variable to track current post ID
    let currentPostId = null;

    /**
     * Opens the comments overlay for a specific post
     * @param {number} postId - The ID of the post to show comments for
     */
    function openCommentsOverlay(postId) {
      currentPostId = postId;
      const overlay = document.getElementById('comments-overlay');
      
      // Show overlay with flex display for centering
      overlay.classList.remove('hidden');
      overlay.classList.add('flex');
      
      // Prevent body scrolling when overlay is open
      document.body.style.overflow = 'hidden';
      
      // Load comments for this post
      loadComments(postId);
    }

    /**
     * Closes the comments overlay
     */
    function closeCommentsOverlay() {
      const overlay = document.getElementById('comments-overlay');
      overlay.classList.add('hidden');
      overlay.classList.remove('flex');
      
      // Restore body scrolling
      document.body.style.overflow = 'auto';
      
      // Clear current post ID
      currentPostId = null;
      
      // Clear comment input
      document.getElementById('comment-input').value = '';
    }

    /**
     * Loads comments for a specific post via AJAX
     * @param {number} postId - The ID of the post to load comments for
     */
    async function loadComments(postId) {
      const container = document.getElementById('comments-container');
      
      try {
        const response = await fetch(`/comments/${postId}`);
        const data = await response.json();
        
        if (data.comments.length === 0) {
          container.innerHTML = `
            <div class="text-center text-gray-500 py-8">
              <svg class="w-12 h-12 mx-auto mb-2 opacity-50" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" 
                      d="M8 12h.01M12 12h.01M16 12h.01M21 12c0 4.418-4.03 8-9 8-1.575 0-3.05-.322-4.313-.894L3 20.25l1.318-3.947C3.488 15.042 3 13.574 3 12c0-4.418 4.03-8 9-8s9 3.582 9 8z">
                </path>
              </svg>
              <p>No comments yet. Be the first to comment!</p>
            </div>
          `;
        } else {
          container.innerHTML = data.comments.map(comment => `
            <div class="flex space-x-3 mb-4 p-3 bg-gray-50 rounded-lg">
              <img src="{{ url_for('static', filename='uploads/profile-pic.jpg') }}" 
                   alt="profile" class="w-8 h-8 rounded-full flex-shrink-0">
              <div class="flex-1">
                <p class="text-sm">
                  <span class="font-semibold text-gray-800">${comment.username}</span>
                  <span class="text-gray-600 ml-2">${comment.content}</span>
                </p>
                <p class="text-xs text-gray-500 mt-1">${comment.timestamp}</p>
              </div>
            </div>
          `).join('');
        }
      } catch (error) {
        console.error('Error loading comments:', error);
        container.innerHTML = `
          <div class="text-center text-red-500 py-8">
            <p>Error loading comments. Please try again.</p>
          </div>
        `;
      }
    }

    /**
     * Submits a new comment for the current post
     */
    async function submitComment() {
      const input = document.getElementById('comment-input');
      const content = input.value.trim();
      
      if (!content) {
        alert('Please enter a comment');
        return;
      }
      
      if (!currentPostId) {
        alert('No post selected');
        return;
      }
      
      try {
        const response = await fetch(`/comments/${currentPostId}`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({ content: content })
        });
        
        if (response.ok) {
          const data = await response.json();
          
          // Clear input
          input.value = '';
          
          // Reload comments to show the new one
          loadComments(currentPostId);
          
          // Update comment count in the post
          const countElement = document.getElementById(`comment-count-${currentPostId}`);
          if (countElement) {
            countElement.textContent = parseInt(countElement.textContent) + 1;
          }
          
        } else {
          const errorData = await response.json();
          alert(errorData.error || 'Error adding comment');
        }
      } catch (error) {
        console.error('Error submitting comment:', error);
        alert('Error submitting comment. Please try again.');
      }
    }

    /**
     * Handles Enter key press in comment input
     * @param {KeyboardEvent} event - The keyboard event
     */
    function handleCommentKeyPress(event) {
      if (event.key === 'Enter') {
        submitComment();
      }
    }

    // Close overlay when clicking outside the content area
    document.getElementById('comments-overlay').addEventListener('click', function(event) {
      if (event.target === this) {
        closeCommentsOverlay();
      }
    });
  </script>

  // ─── State ────────────────────────────────────────────────────────────────
  let currentPostId    = null;

  // ─── Toast ────────────────────────────────────────────────────────────────
  function showToast(msg, type = 'info') {
    const colors = { success: 'bg-gray-900 text-white', error: 'bg-red-500 text-white',
                     warning: 'bg-amber-400 text-white', info: 'bg-gray-700 text-white' };
    const el = document.createElement('div');
    el.className = `pointer-events-auto px-5 py-3 rounded-2xl text-sm font-medium shadow-lg transition-all duration-300 opacity-0 translate-y-2 ${colors[type]||colors.info}`;
    el.textContent = msg;
    document.getElementById('toast-container').appendChild(el);
    requestAnimationFrame(() => el.classList.remove('opacity-0','translate-y-2'));
    setTimeout(() => { el.classList.add('opacity-0','translate-y-2'); setTimeout(()=>el.remove(),300); }, 3200);
  }

  // ─── Block Modal ──────────────────────────────────────────────────────────
  function showBlockModal(reasons) {
    const ul = document.getElementById('block-reasons');
    ul.innerHTML = reasons.map(r =>
      `<li class="flex items-start gap-2">
         <span class="text-red-500 mt-0.5 flex-shrink-0">⛔</span>
         <span>${escHtml(r)}</span>
       </li>`
    ).join('');
    document.getElementById('block-modal').classList.remove('hidden');
    document.getElementById('block-modal').classList.add('flex');
    // Highlight the input in red
    document.getElementById('input-wrapper').classList.add('border-red-400','bg-red-50');
  }

  function closeBlockModal() {
    document.getElementById('block-modal').classList.add('hidden');
    document.getElementById('block-modal').classList.remove('flex');
    const input = document.getElementById('comment-input');
    input.focus(); input.select();
    document.getElementById('input-wrapper').classList.remove('border-red-400','bg-red-50');
  }

  // ─── Warn Modal ───────────────────────────────────────────────────────────
  function showWarnModal(reasons, content) {
    pendingWarnContent = content;
    const ul = document.getElementById('warn-reasons');
    ul.innerHTML = reasons.map(r =>
      `<li class="flex items-start gap-2">
         <span class="text-amber-500 mt-0.5 flex-shrink-0">⚠</span>
         <span>${escHtml(r)}</span>
       </li>`
    ).join('');
    document.getElementById('warn-modal').classList.remove('hidden');
    document.getElementById('warn-modal').classList.add('flex');
  }

  function closeWarnModal() {
    document.getElementById('warn-modal').classList.add('hidden');
    document.getElementById('warn-modal').classList.remove('flex');
    pendingWarnContent = null;
    document.getElementById('comment-input').focus();
  }

  async function forcePostComment() {
    if (!pendingWarnContent || !currentPostId) return;
    closeWarnModal();
    try {
      const res  = await fetch(`/comments/${currentPostId}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content: pendingWarnContent, force_post: true }),
      });
      if (res.ok) {
        document.getElementById('comment-input').value = '';
        resetAnalysisBar();
        updateCommentCount(+1);
        loadComments(currentPostId);
        showToast('Posted (bypassed AI warning)', 'warning');
      } else {
        const d = await res.json();
        showToast(d.error || 'Could not post', 'error');
      }
    } catch { showToast('Network error', 'error'); }
    pendingWarnContent = null;
  }

  // ─── Real-time AI Analysis Bar ────────────────────────────────────────────
  let analysisTimer = null;

  function onCommentInput() {
    const input = document.getElementById('comment-input');
    const bar   = document.getElementById('ai-analysis-bar');
    clearTimeout(analysisTimer);

    if (input.value.trim().length < 2) { resetAnalysisBar(); return; }

    // Show typing indicator immediately
    bar.className = 'mb-2 px-3 py-2 rounded-xl text-xs flex items-center gap-2 transition-all duration-300 bg-gray-50 border border-gray-200';
    document.getElementById('ai-bar-icon').textContent = '⏳';
    document.getElementById('ai-bar-text').textContent = 'Analysing…';
    document.getElementById('ai-bar-badge').textContent = '';
    bar.classList.remove('hidden');

    analysisTimer = setTimeout(async () => {
      if (!currentPostId || !input.value.trim()) return;
      try {
        const res  = await fetch(`/comments/${currentPostId}/analyze`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ content: input.value.trim() }),
        });
        const data = await res.json();
        if (!data) return;
        renderAnalysisBar(data);
      } catch { resetAnalysisBar(); }
    }, 500);   // 500ms debounce
  }

  function renderAnalysisBar(data) {
    const bar    = document.getElementById('ai-analysis-bar');
    const icon   = document.getElementById('ai-bar-icon');
    const text   = document.getElementById('ai-bar-text');
    const badge  = document.getElementById('ai-bar-badge');
    const verdict = data.verdict;
    const reasons = data.reasons || [];

    bar.classList.remove('hidden');

    if (verdict === 'block') {
      bar.className = 'mb-2 px-3 py-2 rounded-xl text-xs flex items-center gap-2 transition-all duration-300 bg-red-50 border border-red-200';
      icon.textContent  = '⛔';
      text.className    = 'flex-1 font-medium text-red-700';
      text.textContent  = reasons[0] || 'Toxic content detected';
      badge.className   = 'px-2 py-0.5 rounded-full text-[10px] font-bold bg-red-500 text-white';
      badge.textContent = 'BLOCKED';
      // Shake the post button
      shakePostBtn();
    } else if (verdict === 'warn') {
      bar.className = 'mb-2 px-3 py-2 rounded-xl text-xs flex items-center gap-2 transition-all duration-300 bg-amber-50 border border-amber-200';
      icon.textContent  = '⚠️';
      text.className    = 'flex-1 font-medium text-amber-700';
      text.textContent  = reasons[0] || 'Potentially flagged';
      badge.className   = 'px-2 py-0.5 rounded-full text-[10px] font-bold bg-amber-400 text-white';
      badge.textContent = 'REVIEW';
    } else {
      bar.className = 'mb-2 px-3 py-2 rounded-xl text-xs flex items-center gap-2 transition-all duration-300 bg-emerald-50 border border-emerald-200';
      const sentiment = data.analysis?.sentiment?.sentiment || 'neutral';
      const emojiMap  = { positive: '😊', negative: '😐', neutral: '😐' };
      icon.textContent  = emojiMap[sentiment] || '😐';
      text.className    = 'flex-1 font-medium text-emerald-700';
      text.textContent  = sentiment === 'positive' ? 'Positive comment' : 'Looks good';
      badge.className   = 'px-2 py-0.5 rounded-full text-[10px] font-bold bg-emerald-500 text-white';
      badge.textContent = '✓ OK';
    }
  }

  function resetAnalysisBar() {
    const bar = document.getElementById('ai-analysis-bar');
    bar.classList.add('hidden');
  }

  function shakePostBtn() {
    const btn = document.getElementById('post-btn');
    btn.style.transition = 'transform 0.07s ease';
    let i = 0;
    const t = setInterval(() => {
      btn.style.transform = i % 2 === 0 ? 'translateX(4px)' : 'translateX(-4px)';
      if (++i > 5) { clearInterval(t); btn.style.transform = 'translateX(0)'; }
    }, 70);
  }

  // ─── Submit Comment ───────────────────────────────────────────────────────
  async function submitComment() {
    const input   = document.getElementById('comment-input');
    const content = input.value.trim();
    if (!content || !currentPostId) return;

    try {
      const res  = await fetch(`/comments/${currentPostId}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ content, force_post: false }),
      });
      const data = await res.json();

      // ── BLOCKED (403) — show block modal, DON'T post ──
      if (res.status === 403) {
        showBlockModal(data.reasons || ['Toxic content detected']);
        return;
      }

      // ── WARNED (422) — show warn modal, user can choose ──
      if (res.status === 422) {
        showWarnModal(data.reasons || ['Content flagged by AI'], content);
        return;
      }

      // ── SUCCESS (201) ──
      if (res.ok) {
        input.value = '';
        resetAnalysisBar();
        updateCommentCount(+1);
        loadComments(currentPostId);
        showToast('Comment posted!', 'success');
      } else {
        showToast(data.error || 'Error posting comment', 'error');
      }
    } catch { showToast('Network error. Please try again.', 'error'); }
  }

  function handleCommentKeyPress(e) {
    if (e.key === 'Enter') submitComment();
  }

  // ─── Load Comments ────────────────────────────────────────────────────────
  async function loadComments(postId) {
    const container = document.getElementById('comments-container');
    container.innerHTML = `<div class="flex justify-center py-10">

    try {
      const url  = aiFilterEnabled ? `/comments/${postId}?filter=ai` : `/comments/${postId}`;

      const data = await res.json();

      if (aiFilterEnabled && data.ai_filtered) {
        document.getElementById('total-comments').textContent    = data.total_original;
        document.getElementById('showing-comments').textContent  = data.total_filtered;
        document.getElementById('filtered-comments').textContent = data.total_original - data.total_filtered;
      }

      if (!data.comments.length) {
        container.innerHTML = `<div class="text-center text-gray-400 py-14 flex flex-col items-center gap-3">
          <svg class="w-10 h-10 opacity-30" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5"
              d="M8 12h.01M12 12h.01M16 12h.01M21 12c0 4.418-4.03 8-9 8-1.575 0-3.05-.322-4.313-.894L3 20.25l1.318-3.947C3.488 15.042 3 13.574 3 12c0-4.418 4.03-8 9-8s9 3.582 9 8z"/>
          </svg>
          <p class="text-sm">${aiFilterEnabled ? 'All comments filtered by AI.' : 'No comments yet. Be the first!'}</p></div>`;
        return;
      }

      container.innerHTML = data.comments.map(c => {
        const approvedBadge = (aiFilterEnabled && c.ai_verdict === 'allow')
          ? `<span class="text-[10px] font-medium text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-full">✓ AI Approved</span>` : '';
        return `
          <div id="comment-${c.id}" class="group flex gap-3 py-3 border-b border-gray-50 last:border-0 overflow-hidden" style="max-height:200px">
            <img src="/static/uploads/profile-pic.jpg" class="w-7 h-7 rounded-full object-cover flex-shrink-0 mt-0.5" alt="${escHtml(c.username)}">
            <div class="flex-1 min-w-0">
              <p class="text-sm leading-snug">
                <span class="font-bold text-gray-900 mr-1.5">${escHtml(c.username)}</span>
                <span class="text-gray-700">${escHtml(c.content)}</span>
              </p>
              <div class="flex items-center gap-2 mt-1">
                <span class="text-[11px] text-gray-400">${c.timestamp}</span>
                ${approvedBadge}
              </div>
            </div>

              class="opacity-0 group-hover:opacity-100 focus:opacity-100 flex-shrink-0 w-6 h-6 flex items-center justify-center rounded-full text-gray-300 hover:text-red-500 hover:bg-red-50 transition self-start mt-0.5">

              </svg>
            </button>
          </div>`;
      }).join('');
    } catch {
      container.innerHTML = `<div class="text-center text-red-400 py-8 text-sm">Failed to load comments.</div>`;
    }
  }

  // ─── Delete Comment ───────────────────────────────────────────────────────
  async function deleteComment(commentId) {
    if (!confirm('Delete this comment?')) return;
    try {
      const res = await fetch(`/comments/${commentId}/delete`, { method: 'DELETE' });
      if (res.ok) {
        const el = document.getElementById(`comment-${commentId}`);
        if (el) {
          el.style.transition = 'all 0.25s ease';
          el.style.opacity = '0';
          el.style.maxHeight = el.offsetHeight + 'px';
          requestAnimationFrame(() => { el.style.maxHeight = '0'; el.style.padding = '0'; });
          setTimeout(() => { el.remove(); updateCommentCount(-1); }, 260);
        }
        showToast('Comment deleted', 'success');
      } else {
        showToast('Could not delete comment', 'error');
      }
    } catch { showToast('Network error', 'error'); }
  }

  // ─── Delete Post ──────────────────────────────────────────────────────────
  function deletePost(postId) {
    const modal = document.getElementById('delete-post-modal');
    modal.classList.remove('hidden');
    modal.classList.add('flex');
    document.getElementById('confirm-delete-post-btn').onclick = async () => {
      try {
        const res = await fetch(`/posts/${postId}/delete`, { method: 'DELETE' });
        if (res.ok) {
          const card = document.getElementById(`post-${postId}`);
          if (card) { card.style.transition = 'all 0.35s'; card.style.opacity = '0'; card.style.transform = 'scale(0.95)'; setTimeout(() => card.remove(), 350); }
          showToast('Post deleted', 'success');
        } else { showToast('Could not delete post', 'error'); }
      } catch { showToast('Network error', 'error'); }
      closeDeletePostModal();
    };
  }

  function closeDeletePostModal() {
    document.getElementById('delete-post-modal').classList.add('hidden');
    document.getElementById('delete-post-modal').classList.remove('flex');
  }

  // ─── Like Toggle ──────────────────────────────────────────────────────────
  function toggleLike(btn, postId) {
    const heart = document.getElementById(`heart-${postId}`);
    const count = document.getElementById(`like-count-${postId}`);
    const liked = heart.getAttribute('fill') === 'currentColor';
    heart.setAttribute('fill', liked ? 'none' : 'currentColor');
    heart.classList.toggle('text-red-500', !liked);
    count.textContent = parseInt(count.textContent) + (liked ? -1 : 1);

  // ─── Overlay Open/Close ───────────────────────────────────────────────────

    document.getElementById('comment-input').value = '';

  // ─── Helpers ──────────────────────────────────────────────────────────────
  function updateCommentCount(delta) {
    if (!currentPostId) return;
    const el = document.getElementById(`comment-count-${currentPostId}`);
    if (el) el.textContent = Math.max(0, parseInt(el.textContent) + delta);

      if (e.target === e.currentTarget) closeDeletePostModal();
    });
